合约 bid
bid(auctionId_, amount) external payable // MetaNFTAuctionBase bool isEthBid = msg.value > 0; if (isEthBid) { // ETH 分支（must match 启动时选 ETH） require(amount == msg.value, "amount mismatch"); bidPrice = _toUsd(msg.value, 18, getPriceInDollar(address(0))); } else { // ERC20 分支（must match 启动时选 USDC） require(amount > 0, "invalid amount"); bidPrice = _toUsd(amount, IERC20Metadata(paymentToken).decimals(), getPriceInDollar(paymentToken)); IERC20(paymentToken).transferFrom(msg.sender, address(this), amount); } require(startingPriceInDollar < bidPrice, "invalid startingPrice"); require(highestBidInDollar < bidPrice, "invalid highestBid"); if (被超价) { 原最高出价者退款（ETH: transfer / ERC20: transfer） } highestBidder = msg.sender; highestBid = ...; highestBidInDollar = bidPrice; emit Bid(...)
⚠️ 常见坑：ETH 拍卖（paymentToken=0x0）不能用 USDC 出价 —— 合约走 ERC20 分支会在零地址调 decimals() 直接 revert（无 message → 前端显示 require(false)）。出价模式必须与拍卖启动时选的支付代币一致。
